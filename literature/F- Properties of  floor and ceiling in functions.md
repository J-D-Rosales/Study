$$
\begin{align}
x-1 < \lfloor x \rfloor \leq x  & \leq \lceil x \rceil < x+1 \\
\left\lceil  \frac{n}{2}  \right\rceil + \left\lfloor  \frac{n}{2}  \right\rfloor    & = n \\
\forall x\geq 0 \land a,b <0 \in \mathbb{N} : \\
\left\lceil  \frac{\left\lceil  \frac{x}{a}  \right\rceil }{b}  \right\rceil   & = \left\lceil  \frac{x}{ab}  \right\rceil  \\
\left\lceil  \frac{\left( \left\lceil  \frac{x}{a}  \right\rceil  \right)}{b}  \right\rceil   & = \left\lceil  \frac{x}{ab}  \right\rceil  \\
\left\lceil  \frac{a}{b}  \right\rceil  & \leq \frac{a+(b-1)}{b} \\
\left\lceil  \frac{a}{b}  \right\rceil  & \geq \frac{a-(b-1)}{b}
\end{align}
$$
